=======users=======
create table public.users (
  id uuid not null default gen_random_uuid (),
  name text not null,
  email text not null,
  password text not null,
  created_at timestamp with time zone null default now(),
  role character varying not null,
  "vehicleId" jsonb[] null default '{}'::jsonb[],
  "searchHistory" jsonb[] null default '{}'::jsonb[],
  address character varying null,
  "watchList" jsonb[] null default '{}'::jsonb[],
  notifications jsonb[] null default '{}'::jsonb[],
  constraint users_pkey primary key (id),
  constraint users_email_key unique (email)
) TABLESPACE pg_default;


======enquiry=======
create table public.enquiry (
  id bigint generated by default as identity not null,
  created_at timestamp with time zone not null default now(),
  user_id uuid null,
  "enquiryDate" character varying null,
  vehicle jsonb null,
  dealer uuid null,
  constraint enquiry_pkey primary key (id),
  constraint enquiry_dealer_fkey foreign KEY (dealer) references users (id),
  constraint enquiry_user_id_fkey foreign KEY (user_id) references users (id)
) TABLESPACE pg_default;
